  summarise(
    n = n(),

    .groups = "drop"
  ) |>
  mutate(

    sd_error = sd_temperature - target_temperature_sd

Spectral Exponents

The generator uses one_over_f_gamma as an amplitude-spectrum exponent: amplitudes scale approximately as 1 / frequency^gamma. Therefore the power spectrum scales approximately as 1 / frequency^(2 * gamma).

The table below estimates the slope of the log power spectrum for each environment. estimated_power_exponent should be close to 2 * one_over_f_gamma; estimated_amplitude_gamma is half of that value and should be close to one_over_f_gamma.

Code 
estimate_spectral_exponent <- function(temperature) {
  y <- temperature - mean(temperature)
  n <- length(y)
  fft_values <- stats::fft(y)
  frequencies <- seq_len(floor(n / 2)) / n
  power <- Mod(fft_values[seq_along(frequencies) + 1L])^2 / n

  keep <- is.finite(power) & power > 0 & frequencies > 0
  fit <- stats::lm(log10(power[keep]) ~ log10(frequencies[keep]))
  power_exponent <- -unname(stats::coef(fit)[[2]])

  tibble(
    estimated_power_exponent = power_exponent,
    estimated_amplitude_gamma = power_exponent / 2,
    spectral_r_squared = summary(fit)$r.squared
  )
}

spectral_summary <- temperatures |>
  group_by(env_series_id, target_gamma, environment_replicate) |>
  summarise(
    estimate_spectral_exponent(temperature),
    .groups = "drop"
  ) |>
  mutate(
    expected_power_exponent = 2 * target_gamma,
    power_exponent_error = estimated_power_exponent - expected_power_exponent,
    amplitude_gamma_error = estimated_amplitude_gamma - target_gamma
  )

Example Spectra

The plot below shows fitted spectra for representative environment series.

Code 
spectrum_data <- temperatures |>
  group_by(target_gamma) |>
  filter(environment_replicate == min(environment_replicate)) |>
  summarise(
    data = list({
      y <- temperature - mean(temperature)
      n <- length(y)
      fft_values <- stats::fft(y)
      frequencies <- seq_len(floor(n / 2)) / n
      power <- Mod(fft_values[seq_along(frequencies) + 1L])^2 / n
      tibble(frequency = frequencies, power = power)
    }),
    .groups = "drop"
  ) |>
  tidyr::unnest(data) |>
  filter(power > 0)

ggplot(spectrum_data, aes(x = frequency, y = power)) +
  geom_point(alpha = 0.08, size = 0.25) +
  geom_smooth(method = "lm", formula = y ~ x, se = FALSE, colour = "red") +
  scale_x_log10() +
  scale_y_log10() +
  facet_wrap(~ target_gamma, ncol = 2) +
  labs(
    x = "Frequency",
    y = "Power"
  ) +
  theme_minimal()
